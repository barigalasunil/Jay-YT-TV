#   [channels]   channel links   -> one entry per channel
# Or choose per line with a prefix:   scene: https://youtu.be/XXXXXXXXXXX
#                                     song: https://youtu.be/XXXXXXXXXXX
#                                     channel: https://www.youtube.com/@SomeChannel
# Optional custom name:               scene: Beach day | https://youtu.be/XXXXXXXXXXX
#
# To REMOVE something: delete its entry in channels.json (removing the line here does nothing).
